function ir_cd2_img( $key ) {
	$m = get_option( 'ir_cd2_imgs', [] ); $id = (int) ( $m[ $key ] ?? 0 );
	return [ 'id' => $id, 'url' => wp_get_attachment_url( $id ), 'alt' => get_post_meta( $id, '_wp_attachment_image_alt', true ) ];
}
function ir_cd2_card( $els, $dark, $o = [] ) {
	return ir_c( array_merge( [ 'bg' => $dark ? 'irbandcard' : 'irsurface', 'border' => $dark ? 'irbandborder' : 'irborder', 'radius' => 22, 'pad' => [ 28, 26 ], 'gap' => 12, 'cls' => 'ir-card' ], $o ), $els );
}
function ir_cd2_head( $eyebrow, $h2, $sub, $dark, $center = true ) {
	$al = $center ? [ 'align' => 'center' ] : [];
	$els = [ ir_eyebrow( $eyebrow, $dark ? 'irbandaccent' : 'iraccent', $al ), ir_h( $h2, 'h2', 'irh2', $dark ? 'irwhite' : 'irtext', $al ) ];
	if ( $sub ) { $els[] = ir_text( '<p>' . $sub . '</p>', $dark ? 'irbandmuted' : 'irmuted', 'irlead', array_merge( $al, [ 'maxw' => 680 ] ) ); }
	return ir_c( [ 'gap' => 14, 'align' => $center ? 'center' : 'flex-start', 'title' => 'Section heading' ], $els );
}
function ir_cd2_cta( $text = 'Get The Playbook for ₦10,000', $o = [] ) {
	return ir_btn( $text . ' →', '#checkout', 'primary', array_merge( [ 'cls' => 'ir-cd-cta' ], $o ) );
}
