function dr_h( $html, $o = [] ) {
	$s = [
		'title' => $html, 'header_size' => $o['tag'] ?? 'h2', 'align' => $o['align'] ?? 'center',
		'title_color' => $o['color'] ?? DR_NAVY,
		'typography_typography' => 'custom', 'typography_font_family' => $o['font'] ?? 'Montserrat',
		'typography_font_weight' => $o['w'] ?? '900',
		'typography_font_size' => ir_px( $o['size'] ?? 40 ), 'typography_font_size_mobile' => ir_px( $o['size_m'] ?? 30 ),
		'typography_line_height' => [ 'unit' => 'em', 'size' => $o['lh'] ?? 1.12, 'sizes' => [] ],
		'typography_letter_spacing' => [ 'unit' => 'px', 'size' => $o['ls'] ?? -0.4, 'sizes' => [] ],
	];
	if ( ! empty( $o['upper'] ) ) { $s['typography_text_transform'] = 'uppercase'; }
	return ir_w( 'heading', $s, [], [ 'cls' => trim( 'dr-h ' . ( $o['cls'] ?? '' ) ) ] );
}

function dr_box( $els, $o = [] ) {
	$s = [
		'content_width' => 'full', 'flex_direction' => $o['dir'] ?? 'column', 'flex_gap' => ir_gap( $o['gap'] ?? 12 ),

		'border_border' => $o['bs'] ?? 'solid', 'border_width' => $o['bw'] ?? ir_dim( 1 ), 'border_color' => $o['bc'] ?? '#E5E7EB',

function dr_svg( $type ) {

function dr_list( $items, $type = 'ok', $cls = '' ) {
	$h = '<ul class="dr-list is-' . $type . ' ' . $cls . '">';
	foreach ( $items as $t ) { $h .= '<li>' . dr_svg( $type ) . '<span>' . $t . '</span></li>'; }
