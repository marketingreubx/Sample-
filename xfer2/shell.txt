function ir_pb_mail( $to, $subject, $inner, $o = [] ) {
	$o = is_array( $o ) ? $o : [];
	$f = 'font-family:-apple-system,BlinkMacSystemFont,\'Segoe UI\',Roboto,Helvetica,Arial,sans-serif';
	$u = is_email( $to ) ? get_user_by( 'email', $to ) : false;
	$name = $o['name'] ?? ( $u && function_exists( 'ir_first_name' ) ? ir_first_name( $u ) : '' );
	$name = ( $name === 'there' ) ? '' : trim( (string) $name );
	/* The email's own heading moves into the header. */
	$title = $o['title'] ?? '';
	if ( $title === '' && preg_match( '#^\s*<h1[^>]*>(.*?)</h1>#is', $inner, $m ) ) {
		$title = trim( html_entity_decode( wp_strip_all_tags( $m[1] ), ENT_QUOTES, 'UTF-8' ) );
		$inner = substr( $inner, strlen( $m[0] ) );
	}
	/* "Welcome to the system, Ada." keeps the name in the title; otherwise the name goes in a small greeting above it. */
	if ( $name !== '' && empty( $o['kicker'] ) && stripos( $title, $name ) === false ) {
		$o['kicker'] = 'Hi ' . $name . ' 👋';
	}
	$kicker = $o['kicker'] ?? '';
	$pre = $o['preheader'] ?? trim( mb_substr( preg_replace( '/\s+/', ' ', html_entity_decode( wp_strip_all_tags( $inner ), ENT_QUOTES, 'UTF-8' ) ), 0, 110 ) );
	/* A light header on purpose: Gmail's dark mode flips light emails cleanly, while white text on a dark header needs hacks that show up as stray bars on iPhone. */
	$hero = '<div style="' . $f . ';font-weight:700;font-size:16px;letter-spacing:.1px;color:#0A1633">Igho Reuben<span style="color:#1660E8">.</span></div>'
		. ( $kicker !== '' ? '<div style="margin-top:30px;' . $f . ';font-size:13px;line-height:1.4;font-weight:700;letter-spacing:.12em;text-transform:uppercase;color:#E8650E">' . esc_html( $kicker ) . '</div>' : '' )
		. ( $title !== '' ? '<div style="margin-top:' . ( $kicker !== '' ? '10' : '30' ) . 'px;' . $f . ';font-size:' . ( mb_strlen( $title ) > 44 ? 24 : 27 ) . 'px;line-height:1.3;font-weight:700;letter-spacing:-.3px;color:#0A1633">' . esc_html( $title ) . '</div>' : '' )
		. ( ! empty( $o['sub'] ) ? '<div style="margin-top:12px;' . $f . ';font-size:16px;line-height:1.6;color:#4F5D78">' . esc_html( $o['sub'] ) . '</div>' : '' );
	$unsub = ! empty( $GLOBALS['ir_mail_unsub'] ) ? ' · <a href="' . esc_url( $GLOBALS['ir_mail_unsub'] ) . '" style="color:#8A97B4">Unsubscribe</a>' : '';
	$html = '<!doctype html><html><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><meta name="color-scheme" content="light dark"><meta name="supported-color-schemes" content="light dark"></head>'
		. '<body style="margin:0;padding:0;background:#F3F6FC">'
		. '<div style="display:none;max-height:0;overflow:hidden;opacity:0;color:#F3F6FC">' . esc_html( $pre ) . str_repeat( '&#8204;&nbsp;', 40 ) . '</div>'
		. '<table role="presentation" width="100%" cellpadding="0" cellspacing="0" border="0" bgcolor="#F3F6FC" style="background:#F3F6FC"><tr><td align="center" style="padding:28px 14px">'
		. '<table role="presentation" width="100%" cellpadding="0" cellspacing="0" border="0" bgcolor="#ffffff" style="max-width:560px;border-radius:20px;overflow:hidden;background:#ffffff;border:1px solid #E6EBF5">'
		. '<tr><td bgcolor="#1660E8" style="height:4px;line-height:4px;font-size:0;background:#1660E8;background-image:linear-gradient(90deg,#1660E8,#4D8BFF 50%,#FF7A1A)">&nbsp;</td></tr>'
		. '<tr><td style="padding:30px 32px 26px">' . $hero . '</td></tr>'
		. '<tr><td style="padding:0 32px"><div style="height:1px;line-height:1px;font-size:0;background:#EEF2F8">&nbsp;</div></td></tr>'
		. '<tr><td style="padding:26px 32px 10px;' . $f . ';font-size:16px;line-height:1.65;color:#3F4A63">' . $inner . '</td></tr>'
		. '<tr><td style="padding:10px 32px 30px;' . $f . '"><div style="border-top:1px solid #EEF2F8;padding-top:20px;font-size:13px;line-height:1.7;color:#8A97B4">Questions? Just reply to this email, or WhatsApp <a href="https://wa.me/2348104699214" style="color:#1660E8;text-decoration:none;font-weight:700">+234 810 469 9214</a>.<br>Igho Reuben · <a href="' . esc_url( home_url( '/' ) ) . '" style="color:#8A97B4">ighoreuben.com</a>' . $unsub . '</div></td></tr>'
		. '</table></td></tr></table></body></html>';
	$h = [ 'Content-Type: text/html; charset=UTF-8' ];
	if ( function_exists( 'ir_mail_from' ) ) { $h[] = 'From: ' . ir_mail_name() . ' <' . ir_mail_from() . '>'; }
	return wp_mail( $to, $subject, $html, $h );
}
