function ir_pb_mail_access( $uid, $set_link = false ) {
	$u = get_userdata( $uid );
	if ( ! $u ) { return; }
	$needs = $set_link || get_user_meta( $uid, 'ir_pb_needs_pass', true );
	$p = 'margin:0 0 14px;font-size:16px;line-height:1.6;color:#33415F';
	$inner = '<p style="' . $p . '">Your Cold DM System is ready. Your scripts, Smart Writer, reply handler and tracker all live in your private dashboard.</p>'
		. '<p style="margin:18px 0 2px;font-size:13px;font-weight:700;letter-spacing:.1em;text-transform:uppercase;color:#8A97B4">Start here</p>'
		. ir_pb_mail_steps( [ [ $needs ? 'Create your password.' : 'Log in.', 'Use the button below.' ], [ 'Save your details in the Writer.', 'Your name, what you do and who you help. It takes a minute.' ], [ 'Send your first 5 DMs.', 'Tap "Send 5 DMs at once", paste 5 handles and send.' ] ] )
		. ir_pb_mail_card( 'Your login email', $u->user_email, 'You log in with this email. Keep it safe.' )
		. ir_pb_btn( $needs ? ir_pb_set_link( $uid ) : ir_pb_url(), $needs ? 'Create password and open' : 'Open the Cold DM System' );
	ir_pb_mail( $u->user_email, 'Welcome to the Cold DM System', $inner, [ 'kicker' => 'Cold DM System · Access ready', 'title' => 'Welcome,', 'preheader' => 'Your Cold DM System is ready. Here is how to send your first 5 DMs today.' ] );
}
