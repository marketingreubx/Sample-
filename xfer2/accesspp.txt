function ir_pp_mail_access( $uid, $set_link = false ) {
	$u = get_userdata( $uid );
	if ( ! $u ) { return; }
	$needs = $set_link || get_user_meta( $uid, 'ir_pb_needs_pass', true );
	$p = 'margin:0 0 14px;font-size:16px;line-height:1.6;color:#33415F';
	$inner = '<p style="' . $p . '">Your AI Profit System is ready. All 3,000+ prompts, the AI tools and your Claude Skills live in your private dashboard.</p>'
		. '<p style="margin:18px 0 2px;font-size:13px;font-weight:700;letter-spacing:.1em;text-transform:uppercase;color:#8A97B4">Start here</p>'
		. ir_pb_mail_steps( [ [ $needs ? 'Create your password.' : 'Log in.', 'Use the button below.' ], [ 'Search any task.', 'Captions, emails, offers, ads, anything you need.' ], [ 'Fill in your details once.', 'Every prompt updates, then copy it into ChatGPT or Claude.' ] ] )
		. ir_pb_mail_card( 'Your login email', $u->user_email, 'You log in with this email. Keep it safe.' )
		. ir_pb_btn( $needs ? ir_pp_set_link( $uid ) : ir_pp_url(), $needs ? 'Create password and open' : 'Open my prompts' );
	ir_pb_mail( $u->user_email, 'Welcome to the AI Profit System', $inner, [ 'kicker' => 'AI Profit System · Access ready', 'title' => 'Welcome,', 'preheader' => 'Your AI Profit System is ready. 3,000+ prompts, AI tools and Claude Skills.' ] );
}
