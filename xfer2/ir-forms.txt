function ir_form_member() {
	return is_user_logged_in() && ( ( function_exists( 'ir_pp_has_access' ) && ir_pp_has_access() ) || ( function_exists( 'ir_pb_has_access' ) && ir_pb_has_access() ) || current_user_can( 'manage_options' ) );
}
function ir_form_get_mine( $uid = 0 ) {
	$f = get_user_meta( $uid ?: get_current_user_id(), 'ir_form', true );
	return is_array( $f ) && ! empty( $f['id'] ) ? $f : null;
}
function ir_form_by_id( $id ) {
	$idx = (array) get_option( 'ir_form_idx', [] );
	$uid = (int) ( $idx[ $id ] ?? 0 );
	$f = $uid ? ir_form_get_mine( $uid ) : null;
	return ( $f && $f['id'] === $id ) ? [ $uid, $f ] : null;
}
function ir_form_clean_color( $c ) { return preg_match( '/^#[0-9a-fA-F]{6}$/', (string) $c ) ? $c : '#1660E8'; }

add_action( 'wp_ajax_ir_form_save', function () {
	check_ajax_referer( 'ir_form', 'n' );
	if ( ! ir_form_member() ) { wp_send_json( [ 'ok' => false, 'err' => 'Please log in again.' ] ); }
	$in = json_decode( wp_unslash( $_POST['data'] ?? '' ), true );
	if ( ! is_array( $in ) ) { wp_send_json( [ 'ok' => false, 'err' => 'Something went wrong. Try again.' ] ); }
	$uid = get_current_user_id();
	$old = ir_form_get_mine( $uid );
	$t = function ( $k, $n = 200 ) use ( $in ) { return mb_substr( sanitize_text_field( (string) ( $in[ $k ] ?? '' ) ), 0, $n ); };
	$qs = [];
	foreach ( array_slice( (array) ( $in['questions'] ?? [] ), 0, 20 ) as $q ) {
		$label = mb_substr( sanitize_text_field( (string) ( $q['label'] ?? '' ) ), 0, 200 );
		if ( $label === '' ) { continue; }
		$type = in_array( $q['type'] ?? '', [ 'short', 'long', 'choice' ], true ) ? $q['type'] : 'short';
		$opts = array_values( array_filter( array_map( function ( $o ) { return mb_substr( sanitize_text_field( (string) $o ), 0, 120 ); }, array_slice( (array) ( $q['options'] ?? [] ), 0, 12 ) ) ) );
		if ( $type === 'choice' && ! $opts ) { $type = 'short'; }
		$qs[] = [ 'label' => $label, 'type' => $type, 'on' => ! empty( $q['on'] ), 'options' => $opts ];
	}
	$logo = (string) ( $in['logo'] ?? '' );
	if ( $logo !== '' && ( strlen( $logo ) > 120000 || ! preg_match( '#^data:image/(png|jpeg);base64,[A-Za-z0-9+/=]+$#', $logo ) ) ) { $logo = $old['logo'] ?? ''; }
	$email = sanitize_email( (string) ( $in['email'] ?? '' ) );
	if ( ! is_email( $email ) ) { wp_send_json( [ 'ok' => false, 'err' => 'Add a valid email for your answers.' ] ); }
	$id = $old['id'] ?? strtolower( wp_generate_password( 8, false, false ) );
	$f = [ 'id' => $id, 'biz' => $t( 'biz', 120 ), 'name' => $t( 'name', 120 ), 'email' => $email, 'color' => ir_form_clean_color( $in['color'] ?? '' ), 'logo' => $logo,
		'title' => $t( 'title', 160 ), 'intro' => mb_substr( sanitize_textarea_field( (string) ( $in['intro'] ?? '' ) ), 0, 600 ), 'thanks' => $t( 'thanks', 300 ), 'questions' => $qs, 'updated' => time() ];
	update_user_meta( $uid, 'ir_form', $f );
	$idx = (array) get_option( 'ir_form_idx', [] );
	$idx[ $id ] = $uid;
	update_option( 'ir_form_idx', $idx, false );
	wp_send_json( [ 'ok' => true, 'form' => $f ] );
} );

add_action( 'wp_ajax_ir_form_subs', function () {
	check_ajax_referer( 'ir_form', 'n' );
	if ( ! ir_form_member() ) { wp_send_json( [ 'ok' => false ] ); }
	$s = get_user_meta( get_current_user_id(), 'ir_form_subs', true );
	wp_send_json( [ 'ok' => true, 'subs' => array_slice( is_array( $s ) ? $s : [], 0, 100 ) ] );
} );

/* Public submit: honeypot, a minimum fill time and a per-IP limit keep spam out. */
function ir_form_submit() {
	$id = sanitize_key( $_POST['f'] ?? '' );
	$found = ir_form_by_id( $id );
	if ( ! $found ) { wp_send_json( [ 'ok' => false, 'err' => 'This form is no longer available.' ] ); }
	list( $uid, $f ) = $found;
	if ( ! empty( $_POST['website'] ) ) { wp_send_json( [ 'ok' => true ] ); }
	$ts = (int) ( $_POST['ts'] ?? 0 );
	if ( ! hash_equals( wp_hash( 'irf' . $ts . $id ), (string) ( $_POST['tk'] ?? '' ) ) || time() - $ts < 3 ) { wp_send_json( [ 'ok' => false, 'err' => 'Please try again.' ] ); }
	$ip = substr( md5( (string) ( $_SERVER['REMOTE_ADDR'] ?? '' ) ), 0, 12 );
	$k = 'irf_ip_' . $ip; $n = (int) get_transient( $k );
	if ( $n >= 8 ) { wp_send_json( [ 'ok' => false, 'err' => 'Too many messages. Please try again later.' ] ); }
	set_transient( $k, $n + 1, HOUR_IN_SECONDS );
	$p = wp_unslash( $_POST );
	$name = mb_substr( sanitize_text_field( (string) ( $p['q_name'] ?? '' ) ), 0, 120 );
	$email = sanitize_email( (string) ( $p['q_email'] ?? '' ) );
	$phone = mb_substr( sanitize_text_field( (string) ( $p['q_phone'] ?? '' ) ), 0, 40 );
	if ( $name === '' || ! is_email( $email ) ) { wp_send_json( [ 'ok' => false, 'err' => 'Please add your name and a valid email.' ] ); }
	$a = [ [ 'Name', $name ], [ 'Email', $email ] ];
	if ( $phone !== '' ) { $a[] = [ 'Phone', $phone ]; }
	foreach ( $f['questions'] as $i => $q ) {
		if ( empty( $q['on'] ) ) { continue; }
		$v = $p[ 'q' . $i ] ?? '';
		$v = is_array( $v ) ? implode( ', ', array_map( 'sanitize_text_field', $v ) ) : ( $q['type'] === 'long' ? sanitize_textarea_field( (string) $v ) : sanitize_text_field( (string) $v ) );
		$a[] = [ $q['label'], mb_substr( $v, 0, 3000 ) ];
	}
	$subs = get_user_meta( $uid, 'ir_form_subs', true );
	$subs = is_array( $subs ) ? $subs : [];
	array_unshift( $subs, [ 'at' => wp_date( 'j M Y, g:i a' ), 'name' => $name, 'email' => $email, 'a' => $a ] );
	update_user_meta( $uid, 'ir_form_subs', array_slice( $subs, 0, 200 ) );
	/* Email the member. Replying goes straight to the person who filled the form. */
	$rows = '';
	foreach ( $a as $r ) {
		$rows .= '<tr><td style="padding:10px 0;border-top:1px solid #E6EBF5;color:#6B7A99;font-size:13px;width:38%;vertical-align:top">' . esc_html( $r[0] ) . '</td><td style="padding:10px 0;border-top:1px solid #E6EBF5;font-size:15px;color:#0A1633;white-space:pre-wrap">' . esc_html( $r[1] !== '' ? $r[1] : '-' ) . '</td></tr>';
	}
	$inner = '<p style="margin:0 0 14px;font-size:16px;line-height:1.6;color:#33415F">' . esc_html( $name ) . ' just filled in your enquiry form. Here are their answers:</p>'
		. '<table role="presentation" width="100%" cellpadding="0" cellspacing="0" border="0" style="margin:6px 0 10px">' . $rows . '</table>'
		. '<p style="margin:16px 0 0;font-size:15px;line-height:1.6;color:#33415F">Tip: reply within the hour. Just hit reply, and your answer goes to ' . esc_html( $email ) . '.</p>';
	$reply = function ( $m ) use ( $email, $name ) { $m['headers'] = (array) $m['headers']; $m['headers'][] = 'Reply-To: ' . str_replace( [ "\r", "\n", '"' ], '', $name ) . ' <' . $email . '>'; return $m; };
	add_filter( 'wp_mail', $reply );
	if ( function_exists( 'ir_pb_mail' ) ) {
		ir_pb_mail( $f['email'], 'New enquiry from ' . $name, $inner, [ 'kicker' => 'New enquiry · ' . ( $f['biz'] ?: 'Your form' ), 'title' => 'New enquiry from ' . $name ] );
	} else {
		wp_mail( $f['email'], 'New enquiry from ' . $name, $inner, [ 'Content-Type: text/html; charset=UTF-8' ] );
	}
	remove_filter( 'wp_mail', $reply );
	wp_send_json( [ 'ok' => true ] );
}
add_action( 'wp_ajax_ir_form_submit', 'ir_form_submit' );
add_action( 'wp_ajax_nopriv_ir_form_submit', 'ir_form_submit' );

/* The public form page: /enquiry/{id}/ */
add_action( 'init', function () {
	$path = trim( (string) wp_parse_url( $_SERVER['REQUEST_URI'] ?? '', PHP_URL_PATH ), '/' );
	if ( ! preg_match( '#^enquiry/([a-z0-9]{8})$#', $path, $m ) ) { return; }
	$found = ir_form_by_id( $m[1] );
	nocache_headers();
	do_action( 'litespeed_control_set_nocache', 'enquiry form' );
	if ( ! $found ) { status_header( 404 ); echo '<!doctype html><meta name="viewport" content="width=device-width,initial-scale=1"><p style="font-family:sans-serif;padding:40px;text-align:center">This form is no longer available.</p>'; exit; }
	status_header( 200 );
	ir_form_render( $found[1] );
	exit;
}, 1 );
