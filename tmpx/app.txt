  if(d&&Array.isArray(d.txs)){S.txs=d.txs;S.settings={...DEF,...(d.settings||{})}}
  else if(d){S.txs=examples()}
  else{try{const c=JSON.parse(localStorage.getItem(LS)||'null');if(c){S.txs=c.txs;S.settings={...DEF,...c.settings}}}catch(e){}toast('Couldn’t reach the server. Showing the last copy on this device.')}
  S.loaded=true;S.mode='wp';render();
  let chain=Promise.resolve();
  const save=()=>{const body=JSON.stringify({txs:S.txs,settings:S.settings});try{localStorage.setItem(LS,body)}catch(e){}
    chain=chain.catch(()=>{}).then(()=>fetch(W.api,{method:'POST',headers:{...hdr,'Content-Type':'application/json'},credentials:'same-origin',body}).then(r=>{if(r.status===401||r.status===403)throw{code:'auth'};if(!r.ok)throw{code:'net'}}));return chain};
  backend={
    saveTx:async(id,tx)=>{S.txs=S.txs.filter(t=>t.id!==id).concat({id,...tx});render();await save()},
    delTx:async id=>{S.txs=S.txs.filter(t=>t.id!==id);render();await save()},
    saveSettings:async st=>{S.settings={...DEF,...st};render();await save()}};

  render();

  if(!db){localBoot();return}
  S.mode='db';
  const col=db.collection('tx'),set=db.doc('settings/main');
  backend={saveTx:(id,tx)=>col.doc(id).set(tx),delTx:id=>col.doc(id).delete(),saveSettings:s=>set.set(s)};
  col.onSnapshot(s=>{S.txs=s.docs.map(d=>({id:d.id,...d.data()}));S.loaded=true;render()},e=>{S.loaded=true;render();toast('Couldn’t load entries. Reload the page to try again.')});
  set.onSnapshot(s=>{if(s.exists)S.settings={...DEF,...s.data()};render()},()=>{});
  try{const u=await window.claude.use('user');if(u){const ok=await u.can('data.write');if(ok===false)S.readOnly=true}}catch(e){}
}
