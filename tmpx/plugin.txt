<?php
/**
 * Plugin Name: Money Desk
 * Description: Private finance tracker at /money/. Only administrators can open it; data is stored in the money_desk_data option.
 */
if ( ! defined( 'ABSPATH' ) ) { exit; }

function md_can() { return current_user_can( 'manage_options' ); }

add_action( 'rest_api_init', function () {
	register_rest_route( 'money-desk/v1', '/data', [
		[
			'methods'             => 'GET',
			'permission_callback' => 'md_can',
			'callback'            => function () {
				nocache_headers();
				$d = get_option( 'money_desk_data', '' );
				$d = $d ? json_decode( $d, true ) : null;
				return is_array( $d ) ? $d : new stdClass();
			},
		],
		[
			'methods'             => 'POST',
			'permission_callback' => 'md_can',
			'callback'            => function ( WP_REST_Request $r ) {
				$b = $r->get_json_params();
				if ( ! is_array( $b ) || ! isset( $b['txs'] ) || ! is_array( $b['txs'] ) ) {
					return new WP_Error( 'md_bad', 'Invalid data.', [ 'status' => 400 ] );
				}
				$j = wp_json_encode( [ 'txs' => array_values( $b['txs'] ), 'settings' => isset( $b['settings'] ) && is_array( $b['settings'] ) ? $b['settings'] : [], 'saved' => time() ] );
				if ( ! $j || strlen( $j ) > 4000000 ) {
					return new WP_Error( 'md_big', 'Too much data.', [ 'status' => 413 ] );
				}
				$prev = get_option( 'money_desk_data', '' );
				if ( $prev ) { update_option( 'money_desk_data_prev', $prev, false ); }
				update_option( 'money_desk_data', $j, false );
				return [ 'ok' => true ];
			},
		],
	] );
} );

add_action( 'template_redirect', function () {
	$path = trim( (string) wp_parse_url( $_SERVER['REQUEST_URI'] ?? '', PHP_URL_PATH ), '/' );
	if ( $path !== 'money' ) { return; }
	do_action( 'litespeed_control_set_nocache', 'money desk is private' );
	nocache_headers();
	header( 'X-Robots-Tag: noindex, nofollow' );
	if ( ! is_user_logged_in() ) { auth_redirect(); }
	if ( ! md_can() ) { wp_die( 'This page is private.', 'Private', [ 'response' => 403 ] ); }
	$app = WP_CONTENT_DIR . '/money-desk/app.html';
	if ( ! is_readable( $app ) ) { wp_die( 'Money Desk files are missing.' ); }
	status_header( 200 );
	header( 'Content-Type: text/html; charset=utf-8' );
	$cfg = wp_json_encode( [ 'api' => esc_url_raw( rest_url( 'money-desk/v1/data' ) ), 'nonce' => wp_create_nonce( 'wp_rest' ) ] );
	$icon = 'data:image/svg+xml,' . rawurlencode( '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 64 64"><rect width="64" height="64" rx="14" fill="#12151C"/><rect x="14" y="34" width="9" height="16" rx="2" fill="#2a78d6"/><rect x="27.5" y="22" width="9" height="28" rx="2" fill="#2a78d6"/><rect x="41" y="28" width="9" height="22" rx="2" fill="#eb6834"/></svg>' );
	echo '<!doctype html><html lang="en"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1,viewport-fit=cover"><meta name="robots" content="noindex,nofollow"><meta name="apple-mobile-web-app-capable" content="yes"><meta name="mobile-web-app-capable" content="yes"><meta name="apple-mobile-web-app-title" content="Money Desk"><meta name="theme-color" content="#ECEEF2" media="(prefers-color-scheme: light)"><meta name="theme-color" content="#0E1014" media="(prefers-color-scheme: dark)">';
	echo '<link rel="icon" href="' . esc_attr( $icon ) . '"><link rel="apple-touch-icon" href="' . esc_attr( $icon ) . '">';
	echo '<style>html{color-scheme:light}:root{padding-top:env(safe-area-inset-top,0px);padding-bottom:env(safe-area-inset-bottom,0px)}body{margin:0}img{max-width:100%}[hidden]{display:none!important}</style>';
	echo '<script>window.MD_WP=' . $cfg . ';</script></head><body>';
	readfile( $app );
	echo '</body></html>';
	exit;
}, 0 );
