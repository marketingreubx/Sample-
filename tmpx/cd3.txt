function dr_h( $html, $o = [] ) {
	return ir_w( 'heading', [
		'title' => $html, 'header_size' => $o['tag'] ?? 'h2', 'align' => $o['align'] ?? 'left',
		'title_color' => $o['color'] ?? DR_INK,
		'typography_typography' => 'custom', 'typography_font_family' => 'Source Serif 4',
		'typography_font_weight' => $o['w'] ?? '700',
		'typography_font_size' => ir_px( $o['size'] ?? 38 ), 'typography_font_size_mobile' => ir_px( $o['size_m'] ?? 30 ),
		'typography_line_height' => [ 'unit' => 'em', 'size' => $o['lh'] ?? 1.15, 'sizes' => [] ],
		'typography_letter_spacing' => [ 'unit' => 'px', 'size' => $o['ls'] ?? -0.4, 'sizes' => [] ],
	], [], [ 'cls' => trim( 'dr-h ' . ( $o['cls'] ?? '' ) ) ] );
}
function dr_p( $html, $o = [] ) {
	return ir_w( 'text-editor', [
		'editor' => $html, 'align' => $o['align'] ?? 'left', 'text_color' => $o['color'] ?? DR_BODY,
		'typography_typography' => 'custom', 'typography_font_family' => $o['font'] ?? 'Source Serif 4',
		'typography_font_weight' => $o['w'] ?? '400',
		'typography_font_size' => ir_px( $o['size'] ?? 21 ), 'typography_font_size_mobile' => ir_px( $o['size_m'] ?? 20 ),
		'typography_line_height' => [ 'unit' => 'em', 'size' => $o['lh'] ?? 1.65, 'sizes' => [] ],
	], [], [ 'cls' => trim( 'dr-p ' . ( $o['cls'] ?? '' ) ) ] );
}
function dr_label( $t, $align = 'left' ) {
	return dr_p( '<p>' . $t . '</p>', [ 'font' => 'Archivo', 'w' => '600', 'size' => 15, 'size_m' => 15, 'lh' => 1.4, 'align' => $align, 'color' => DR_MUTED, 'cls' => 'dr-label' ] );
}
function dr_sec( $title, $bg, $els, $o = [] ) {
	return [ 'id' => ir_id(), 'elType' => 'container', 'isInner' => false, 'elements' => $els, 'settings' => [
		'content_width' => 'boxed', 'boxed_width' => ir_px( $o['w'] ?? 720 ), 'flex_direction' => 'column',
		'flex_align_items' => $o['align'] ?? 'stretch',
		'flex_gap' => ir_gap( $o['gap'] ?? 22 ), 'flex_gap_mobile' => ir_gap( $o['gap_m'] ?? 18 ),
		'padding' => ir_dim( $o['pt'] ?? 80, 22 ), 'padding_mobile' => ir_dim( $o['ptm'] ?? 56, 20 ),
		'background_background' => 'classic', 'background_color' => $bg,
		'css_classes' => trim( 'dr-sec ' . ( $o['cls'] ?? '' ) ), '_title' => $title,
	] ];
}
function dr_list( $items, $cls = '' ) {
	$h = '<ul class="dr-list ' . $cls . '">';
	foreach ( $items as $t ) { $h .= '<li>' . $t . '</li>'; }
	return ir_html( $h . '</ul>' );
}
function dr_cta( $center = false, $label = 'Get the playbook for ₦4,500' ) {
	return ir_html( '<div class="dr-ctaw' . ( $center ? ' is-c' : '' ) . '"><a href="#checkout" class="dr-cta">' . esc_html( $label ) . '</a><p class="dr-cta-note">Paystack checkout. 14-day money-back guarantee. Access sent on WhatsApp.</p></div>' );
}
