add_action( 'wp_footer', function () {
	if ( ! is_front_page() ) { return; }
	?>
<script id="ir-hero-rotator">
(function(){
  var el=document.querySelector('.elementor-element-9f35954 .ir-accent');
  if(!el)return;
  var words=[['more customers.','irw-c1'],['more sales.','irw-c2'],['more visibility.','irw-c3'],['more bookings.','irw-c4'],['more leads.','irw-c5']];
  el.classList.add('irw','irw-c1');
  if(window.matchMedia&&matchMedia('(prefers-reduced-motion: reduce)').matches)return;
  var i=0;
  setInterval(function(){
    if(document.hidden)return;
    i=(i+1)%words.length;
    el.classList.add('irw-out');
    setTimeout(function(){
      el.textContent=words[i][0];
      el.className='ir-accent irw '+words[i][1]+' irw-in';
      void el.offsetWidth;
      el.classList.remove('irw-in');
    },400);
  },2600);
})();
</script>
	<?php
}, 30 );
