function ir_ty_page( $slug, $title, $html ) {
	$p  = get_page_by_path( $slug );
	$id = $p ? $p->ID : wp_insert_post( [ 'post_type' => 'page', 'post_status' => 'publish', 'post_title' => $title, 'post_name' => $slug ] );
	update_post_meta( $id, '_wp_page_template', 'elementor_canvas' );
	$c            = ir_c( [ 'title' => 'Thank you page (pixel Purchase event inside)' ], [ ir_html( $html ) ] );
	$c['isInner'] = false;
	ir_save( $id, [ $c ], [ 'hide_title' => 'yes' ] );
	return $id;
}
