<?php
/** One-off: AmpliQ-style restyle in Ighoreuben brand colours. Styling only, no content changes. */
if ( ! defined( 'ABSPATH' ) || ! defined( 'IR_RESTYLE' ) ) { return; }
$out = [];

/* 1. Kit: colours + typography. */
$kit = get_post_meta( 205, '_elementor_page_settings', true );
update_post_meta( 205, '_ir_kit_backup_before_restyle', $kit );
$colors = [
	'irbg' => '#F8FAFF', 'irsurface' => '#FFFFFF', 'irband' => '#071233', 'irtext' => '#0A1633', 'irmuted' => '#4F5D78',
	'iraccent' => '#1660E8', 'irhover' => '#0F4ACB', 'irsoft' => '#EEF3FF', 'irborder' => '#E5EBF6', 'iralt' => '#F1F5FF',
	'irbandcard' => '#12204A', 'irbandborder' => '#22346A', 'irbandaccent' => '#FF9A52', 'irbandmuted' => '#A9B6D0',
];
foreach ( $kit['custom_colors'] as &$c ) { if ( isset( $colors[ $c['_id'] ] ) ) { $c['color'] = $colors[ $c['_id'] ]; } }
unset( $c );
$have = array_column( $kit['custom_colors'], '_id' );
if ( ! in_array( 'irorange', $have, true ) ) { $kit['custom_colors'][] = [ '_id' => 'irorange', 'title' => 'IR Orange', 'color' => '#FF7A1A' ]; }
$ls = [ 'irh1' => -2, 'irh2' => -1.3, 'irh3' => -0.5, 'irh4' => -0.3, 'irlogo' => -0.6, 'irstat' => -1.6 ];
foreach ( $kit['custom_typography'] as &$t ) {
	if ( ! empty( $t['typography_font_family'] ) ) { $t['typography_font_family'] = 'Host Grotesk'; }
	if ( isset( $ls[ $t['_id'] ] ) ) { $t['typography_letter_spacing'] = [ 'unit' => 'px', 'size' => $ls[ $t['_id'] ], 'sizes' => [] ]; }
}
unset( $t );

/* 2. Mark plain dark bands (no photo background) so they get the gradient. */
$bands = 0;
foreach ( [ 2535, 2548, 2551, 2561, 2573, 2580, 812, 2618, 2526 ] as $pid ) {
	$raw = get_post_meta( $pid, '_elementor_data', true );
	$d   = json_decode( $raw, true );
	if ( ! is_array( $d ) ) { continue; }
	update_post_meta( $pid, '_ir_backup_before_restyle', wp_slash( $raw ) );
	$ch = false;
	foreach ( $d as &$s ) {
		$st  = &$s['settings'];
		$cls = $st['css_classes'] ?? '';
		$bg  = $st['__globals__']['background_color'] ?? '';
		$img = ! empty( $st['background_image']['url'] );
		if ( strpos( $bg, 'irband' ) !== false && ! $img && strpos( $cls, 'irx-band' ) === false ) { $st['css_classes'] = trim( $cls . ' irx-band' ); $ch = true; $bands++; }
		if ( strpos( $cls, 'ir-dots' ) !== false && strpos( $cls, 'irx-hero' ) === false ) { $st['css_classes'] = trim( $st['css_classes'] . ' irx-hero' ); $ch = true; }
		unset( $st );
	}
	unset( $s );
	if ( $ch ) { update_post_meta( $pid, '_elementor_data', wp_slash( wp_json_encode( $d ) ) ); delete_post_meta( $pid, '_elementor_element_cache' ); }
}
$out['bands'] = $bands;

/* 3. CSS. Scoped away from canvas landing pages (Cold DM, prompts, thank-you). */
$B = 'body:not(.elementor-template-canvas)';
$css = <<<CSS
/* AmpliQ style */
.elementor-kit-205{--e-global-color-irorange:#FF7A1A}
html[data-theme="dark"] .elementor-kit-205{--e-global-color-irorange:#FF8A3D;--e-global-color-irbandaccent:#FF9A52}
{$B}{font-family:'Host Grotesk',system-ui,-apple-system,sans-serif}
{$B} .elementor-widget-text-editor,{$B} .elementor-button,{$B} .elementor-icon-list-text{font-family:'Host Grotesk',system-ui,-apple-system,sans-serif}
{$B} .elementor-heading-title{font-feature-settings:'ss01'}
/* accent words: blue into orange */
{$B} .elementor-heading-title .ir-accent{background:linear-gradient(90deg,#1660E8 0%,#3D7BFF 55%,#FF7A1A 100%);-webkit-background-clip:text;background-clip:text;-webkit-text-fill-color:transparent;color:transparent}
html[data-theme="dark"] {$B} .elementor-heading-title .ir-accent,{$B} .ir-on-band .elementor-heading-title .ir-accent{background:linear-gradient(90deg,#6EA8FE 0%,#8FB8FF 50%,#FF9A52 100%);-webkit-background-clip:text;background-clip:text}
/* eyebrow labels: orange */
{$B} .ir-eyebrow-line .elementor-heading-title{color:var(--e-global-color-irorange)!important;letter-spacing:.16em}
{$B} .ir-eyebrow-line .elementor-heading-title::before{background:var(--e-global-color-irorange)!important}
html:not([data-theme="dark"]) {$B} .ir-eyebrow-line .elementor-heading-title{color:#E5620A!important}
/* glass header */
.ir-header::before{background:rgba(255,255,255,.62)!important;-webkit-backdrop-filter:blur(18px) saturate(160%);backdrop-filter:blur(18px) saturate(160%);box-shadow:inset 0 1px 0 rgba(255,255,255,.8),0 1px 0 rgba(10,22,51,.06),0 10px 30px rgba(10,22,51,.06)}
html[data-theme="dark"] .ir-header::before{background:rgba(7,15,31,.62)!important;box-shadow:0 1px 0 rgba(255,255,255,.06),0 10px 30px rgba(0,0,0,.3)}
/* pastel mesh hero (light mode) */
html:not([data-theme="dark"]) {$B} .irx-hero{background:radial-gradient(55% 50% at 8% 18%,rgba(110,200,255,.45),transparent 70%),radial-gradient(45% 45% at 92% 12%,rgba(130,150,255,.38),transparent 70%),radial-gradient(55% 55% at 88% 92%,rgba(255,168,110,.42),transparent 72%),radial-gradient(45% 40% at 20% 95%,rgba(150,225,255,.35),transparent 70%),linear-gradient(140deg,#E2F5FF 0%,#E9EEFF 45%,#FFF0E4 100%)!important}
/* deep gradient bands */
{$B} .irx-band{background:radial-gradient(90% 70% at 100% 100%,rgba(29,94,255,.55),rgba(29,94,255,0) 60%),radial-gradient(55% 50% at 0% 0%,rgba(255,122,26,.20),rgba(255,122,26,0) 60%),linear-gradient(160deg,#0A1A4A 0%,#050B22 100%)!important}

/* cards */
{$B} .ir-card{border-radius:24px!important;border-color:rgba(10,22,51,.07)!important;box-shadow:0 1px 2px rgba(10,22,51,.04),0 14px 40px rgba(10,22,51,.07)!important;transition:transform .25s ease,box-shadow .25s ease}
@media(hover:hover){{$B} .ir-card:hover{transform:translateY(-3px);box-shadow:0 2px 4px rgba(10,22,51,.05),0 22px 50px rgba(10,22,51,.10)!important}}
html[data-theme="dark"] {$B} .ir-card{border-color:rgba(255,255,255,.08)!important;box-shadow:0 14px 40px rgba(0,0,0,.35)!important}
/* chips + icon tiles */
{$B} .ir-chip{border-radius:999px!important;background:rgba(255,255,255,.72)!important;border:1px solid rgba(10,22,51,.08)!important;-webkit-backdrop-filter:blur(8px);backdrop-filter:blur(8px)}
html[data-theme="dark"] {$B} .ir-chip,{$B} .ir-on-band .ir-chip{background:rgba(255,255,255,.07)!important;border-color:rgba(255,255,255,.14)!important}
{$B} .ir-icon-sq{border-radius:14px!important}
/* buttons: softer shadow on the blue ones */
{$B} .elementor-widget-button .elementor-button{transition:transform .2s ease,box-shadow .2s ease,background-color .2s ease}
@media(hover:hover){{$B} .elementor-widget-button .elementor-button:hover{transform:translateY(-1px)}}
/* footer */
.elementor-location-footer .irx-band,.elementor-location-footer .ir-on-band{background:radial-gradient(80% 90% at 100% 100%,rgba(29,94,255,.5),rgba(29,94,255,0) 60%),radial-gradient(50% 60% at 0% 0%,rgba(255,122,26,.16),rgba(255,122,26,0) 60%),linear-gradient(160deg,#0A1A4A 0%,#050B22 100%)!important}
CSS;

/* add or replace the block in kit custom CSS */
$cur = $kit['custom_css'] ?? '';
$cur = preg_replace( '#/\* AmpliQ style \*/.*$#s', '', $cur );
$kit['custom_css'] = rtrim( $cur ) . "\n" . $css . "\n";
update_post_meta( 205, '_elementor_page_settings', $kit );

/* 4. Load Host Grotesk on the front end. */
$mu = WP_CONTENT_DIR . '/mu-plugins/ir-fonts.php';
file_put_contents( $mu, "<?php\n/** Plugin Name: IR Fonts\n * Description: Loads Host Grotesk from Google Fonts. */\nif ( ! defined( 'ABSPATH' ) ) { exit; }\nadd_action( 'wp_head', function () {\n\techo '<link rel=\"preconnect\" href=\"https://fonts.googleapis.com\"><link rel=\"preconnect\" href=\"https://fonts.gstatic.com\" crossorigin><link rel=\"stylesheet\" href=\"https://fonts.googleapis.com/css2?family=Host+Grotesk:wght@300..800&display=swap\">' . \"\\n\";\n}, 2 );\n" );

\Elementor\Plugin::$instance->files_manager->clear_cache();
do_action( 'litespeed_purge_all' );
$GLOBALS['ir_restyle_out'] = $out;
