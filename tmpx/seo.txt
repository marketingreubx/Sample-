function ir_seo_is_noindex() {
	if ( is_search() || is_404() || is_author() || is_attachment() || is_date() || is_tag() ) { return true; }
	if ( function_exists( 'is_cart' ) && ( is_cart() || is_checkout() || is_account_page() ) ) { return true; }
	return is_singular() && in_array( (int) get_queried_object_id(), ir_seo_noindex_ids(), true );
}

function ir_seo_current() {
	if ( ! is_singular() && ! is_front_page() ) { return null; }
	$id  = (int) ( is_front_page() ? get_option( 'page_on_front' ) : get_queried_object_id() );
	$map = ir_seo_pages();
	if ( isset( $map[ $id ] ) ) { return [ 'id' => $id, 'title' => $map[ $id ][0], 'desc' => $map[ $id ][1], 'service' => $map[ $id ][2] ?? '' ]; }
	$p = get_post( $id );
	if ( ! $p ) { return null; }
	$text = $p->post_excerpt ?: $p->post_content;
	$text = trim( preg_replace( '/\s+/', ' ', wp_strip_all_tags( strip_shortcodes( $text ) ) ) );
	$desc = $text ? wp_html_excerpt( $text, 155, '…' ) : '';
	return [ 'id' => $id, 'title' => get_the_title( $id ) . ' | Igho Reuben', 'desc' => $desc, 'service' => '' ];
}

add_filter( 'pre_get_document_title', function ( $t ) {
	$c = ir_seo_current();
	return $c ? $c['title'] : $t;
}, 99 );

function ir_seo_image( $id ) {
	if ( $id && has_post_thumbnail( $id ) ) { return wp_get_attachment_image_url( get_post_thumbnail_id( $id ), 'large' ); }
	$work = [ 2573 => 'georgia', 2580 => 'colddm2', 2069 => 'colddm2' ];
	$map  = get_option( 'ir_work_map', [] );
	if ( isset( $work[ $id ], $map[ $work[ $id ] ] ) ) { return wp_get_attachment_url( (int) $map[ $work[ $id ] ] ); }
	return wp_get_attachment_url( 2635 ) ?: '';
}

/* Core sitemap: drop noindexed pages, users and low-value taxonomies. */
add_filter( 'wp_sitemaps_posts_query_args', function ( $args ) {
	$args['post__not_in'] = array_merge( $args['post__not_in'] ?? [], ir_seo_noindex_ids() );
	return $args;
} );
add_filter( 'wp_sitemaps_add_provider', function ( $provider, $name ) { return $name === 'users' ? false : $provider; }, 10, 2 );
add_filter( 'wp_sitemaps_taxonomies', function ( $tax ) { return array_intersect_key( $tax, array_flip( [ 'category', 'product_cat' ] ) ); } );
add_filter( 'wp_sitemaps_post_types', function ( $types ) { unset( $types['attachment'] ); return $types; } );
