<?php
/**
 * Plugin Name: IR Speed
 * Description: Stops Tutor LMS and WooCommerce product-page files loading on pages that do not use them.
 */
if ( ! defined( 'ABSPATH' ) ) { exit; }

/** Pages where Tutor LMS must load everything (courses, login, registration, dashboard, academy). */
function ir_speed_needs_tutor() {
	if ( is_admin() || ( function_exists( 'is_account_page' ) && is_account_page() ) ) { return true; }
	if ( is_singular( [ 'courses', 'lesson', 'tutor_quiz', 'tutor_assignments', 'course-bundle' ] ) || is_post_type_archive( 'courses' ) || is_tax( [ 'course-category', 'course-tag' ] ) ) { return true; }
	return is_page( [ 1573, 1574, 1575, 1582, 1647, 2475, 2618 ] );
}

add_action( 'wp_enqueue_scripts', function () {
	if ( is_admin() || isset( $_GET['elementor-preview'] ) ) { return; }

	if ( ! ir_speed_needs_tutor() ) {
		global $wp_scripts, $wp_styles;
		foreach ( (array) $wp_scripts->queue as $h ) {
			if ( strpos( $h, 'tutor' ) === 0 ) { wp_dequeue_script( $h ); }
		}
		foreach ( (array) $wp_styles->queue as $h ) {
			if ( strpos( $h, 'tutor' ) === 0 || $h === 'enrollment-frontend-css' ) { wp_dequeue_style( $h ); }
		}
	}

	if ( ! function_exists( 'is_product' ) || ! is_product() ) {
		foreach ( [ 'wc-single-product', 'wc-add-to-cart-variation', 'wc-zoom', 'wc-flexslider', 'wc-photoswipe', 'wc-photoswipe-ui-default', 'zoom', 'flexslider', 'photoswipe', 'photoswipe-ui-default' ] as $h ) { wp_dequeue_script( $h ); }
		foreach ( [ 'photoswipe', 'photoswipe-default-skin' ] as $h ) { wp_dequeue_style( $h ); }
	}
}, 9999 );
